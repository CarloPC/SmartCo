// UPDATED FIRESTORE RULES - Copy these to your Firebase Console
// Includes Emergency, Fake Report, User Suspension, Notification rules
// and FIXED Food Aid rules so residents can view distributions
// + NEW Announcements collection rules for Community Board (Events feed)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function isAuthenticated() {
      return request.auth != null;
    }

    // admin OR barangay_official
    function isAdmin() {
      return isAuthenticated() &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role in ['admin', 'barangay_official'];
    }

    // admin only
    function isSuperAdmin() {
      return isAuthenticated() &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    function isOwner(userId) {
      return isAuthenticated() && request.auth.uid == userId;
    }

    // ─── Users ──────────────────────────────────────────────────────────────
    match /users/{userId} {
      allow read: if isAuthenticated() && (isOwner(userId) || isAdmin());
      allow create: if isAuthenticated() && isOwner(userId);

      // Users can update their own profile (but not their role)
      allow update: if isAuthenticated() && isOwner(userId) &&
                      (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['role']));

      // Admins and barangay_officials can update users (e.g. suspension fields) — but NOT role
      allow update: if isAdmin() &&
                      (!request.resource.data.diff(resource.data).affectedKeys().hasAny(['role']));

      // Only super admin (admin role) can change roles
      allow update: if isSuperAdmin();

      allow list: if isAdmin();
    }

    // ─── Health Records ──────────────────────────────────────────────────────
    match /healthRecords/{recordId} {
      allow read: if isAuthenticated() && (
        resource.data.userId == request.auth.uid || isAdmin()
      );
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid;
      allow update: if isAdmin();
      allow delete: if isAuthenticated() && (
        (resource.data.userId == request.auth.uid && resource.data.approvalStatus == 'pending') ||
        isAdmin()
      );
    }

    // ─── Food Aid ────────────────────────────────────────────────────────────
    // FIX: All authenticated users can READ food aid distributions.
    // Food aid schedules are community/public information — residents need to see
    // upcoming distributions, progress, and schedule posted by barangay officials.
    match /foodAid/{scheduleId} {
      allow read: if isAuthenticated();
      allow create: if isAuthenticated() && request.resource.data.createdBy == request.auth.uid;
      allow update: if isAuthenticated() && (
        (resource.data.createdBy == request.auth.uid && resource.data.approvalStatus == 'pending') ||
        isAdmin()
      );
      allow delete: if isAuthenticated() && (
        (resource.data.createdBy == request.auth.uid && resource.data.approvalStatus == 'pending') ||
        isAdmin()
      );
    }

    // ─── Events ──────────────────────────────────────────────────────────────
    match /events/{eventId} {
      allow read: if isAuthenticated() && (
        resource.data.createdBy == request.auth.uid || isAdmin()
      );
      allow create: if isAuthenticated() && request.resource.data.createdBy == request.auth.uid;
      allow update: if isAuthenticated() && (
        (resource.data.createdBy == request.auth.uid && resource.data.approvalStatus == 'pending') ||
        isAdmin()
      );
      allow delete: if isAuthenticated() && (
        (resource.data.createdBy == request.auth.uid && resource.data.approvalStatus == 'pending') ||
        isAdmin()
      );
    }

    // ─── Announcements (Community Board) ─────────────────────────────────────
    // Barangay officials/admins can post announcements and events to the feed.
    // All authenticated residents can READ, LIKE (update likes array), and COMMENT.
    // Only officials can create top-level posts.
    // Officials can reply to comments (update the comment's replies array).
    match /announcements/{postId} {
      // All authenticated users can read posts
      allow read: if isAuthenticated();

      // Only officials/admins can create posts
      allow create: if isAdmin();

      // All authenticated users can update (for toggling likes & incrementing commentCount)
      // Officials can also edit their own posts
      allow update: if isAuthenticated();

      // Officials or the original author can delete a post
      allow delete: if isAdmin() ||
                       (isAuthenticated() && resource.data.authorId == request.auth.uid);

      // ── Comments subcollection ─────────────────────────────────────────────
      match /comments/{commentId} {
        // All authenticated users can read comments
        allow read: if isAuthenticated();

        // All authenticated users can comment
        allow create: if isAuthenticated();

        // Any authenticated user can update a comment
        // (residents add replies via arrayUnion — officials use this to reply)
        allow update: if isAuthenticated();

        // Comment author or officials can delete
        allow delete: if isAuthenticated() && (
          resource.data.authorId == request.auth.uid || isAdmin()
        );
      }
    }

    // ─── Emergencies ─────────────────────────────────────────────────────────
    // Any authenticated resident can create; admins can read/update/delete all.
    match /emergencies/{emergencyId} {
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid;
      allow read:   if isAuthenticated() && (
        resource.data.userId == request.auth.uid || isAdmin()
      );
      allow list:   if isAdmin();
      allow update: if isAdmin();
      allow delete: if isSuperAdmin();
    }

    // ─── Admin Access Requests ───────────────────────────────────────────────
    // Any authenticated user can submit a request (must own the userId field).
    // Users can read their own requests to check status.
    // Only admins can read all requests and approve/reject (update status).
    match /adminRequests/{requestId} {
      allow create: if isAuthenticated() && request.resource.data.userId == request.auth.uid;
      allow read:   if isAuthenticated() && (
        resource.data.userId == request.auth.uid || isAdmin()
      );
      allow list:   if isAdmin();
      allow update: if isAdmin();
      allow delete: if isSuperAdmin();
    }

    // ─── Notifications ───────────────────────────────────────────────────────
    // Any authenticated user can CREATE a notification for any userId.
    // This is required for residents to notify admins when reporting emergencies.
    match /notifications/{notificationId} {
      allow read:   if isAuthenticated() && resource.data.userId == request.auth.uid;
      allow create: if isAuthenticated();
      allow update: if isAuthenticated() && resource.data.userId == request.auth.uid;
      allow delete: if isAuthenticated() && resource.data.userId == request.auth.uid;
    }
  }
}
